CREATE TABLE "UP07".Role (
    ID SERIAL PRIMARY KEY,
    Name VARCHAR(30) NOT NULL,
    Permissions VARCHAR(50)[] NOT NULL
);

CREATE TABLE "UP07".Status (
    ID SERIAL PRIMARY KEY,
    Name VARCHAR(50) NOT NULL,
    Description VARCHAR(300) NOT NULL
);

CREATE TABLE  "UP07".Category (
    ID SERIAL PRIMARY KEY,
    Code VARCHAR(5) NOT NULL,
    Name VARCHAR(50) NOT NULL,
    Description VARCHAR(300)
);

CREATE TABLE  "UP07"."Authorization" (
    ID SERIAL PRIMARY KEY,
    Login VARCHAR(30) NOT NULL,
    Password VARCHAR(100) NOT NULL
);

CREATE TABLE  "UP07"."User" (
    ID SERIAL PRIMARY KEY,
    Role_ID INTEGER NOT NULL REFERENCES Role(ID),
    Auth_ID INTEGER NOT NULL REFERENCES "Authorization"(ID),
    Name VARCHAR(40) NOT NULL,
    Surname VARCHAR(40) NOT NULL,
    Otchestvo VARCHAR(40),
    Phone_Number VARCHAR(11),
    Email VARCHAR(30)
);

-- Таблица Учащихся
CREATE TABLE  "UP07".Student (
    ID SERIAL PRIMARY KEY,
    Name VARCHAR(40) NOT NULL,
    Surname VARCHAR(40) NOT NULL,
    Otchestvo VARCHAR(40),
    Date_of_Birth DATE NOT NULL,
    Passport VARCHAR(10) NOT NULL,
    Phone_Number VARCHAR(11),
    Email VARCHAR(30)
);

-- Таблица Инструкторов (связана с Пользователем)
CREATE TABLE  "UP07".Instructor (
    ID SERIAL PRIMARY KEY,
    User_ID INTEGER NOT NULL REFERENCES "User"(ID),
    Experience INTEGER NOT NULL,
    Qualification VARCHAR(100) NOT NULL
);

-- Таблица Учебных групп (связана с Категорией)
CREATE TABLE  "UP07"."Group" (
    ID SERIAL PRIMARY KEY,
    Number VARCHAR(20) NOT NULL,
    Category_ID INTEGER NOT NULL REFERENCES Category(ID),
    Start_Date DATE NOT NULL,
    End_Date DATE,
    Status VARCHAR(20) NOT NULL
);

-- Промежуточная таблица Зачисления (связь М:М между Учащимся и Группой)
CREATE TABLE  "UP07".Enrollment (
    ID SERIAL PRIMARY KEY,
    Student_ID INTEGER NOT NULL REFERENCES Student(ID),
    Group_ID INTEGER NOT NULL REFERENCES "Group"(ID),
    Enrollment_Date DATE NOT NULL
);

-- Таблица Транспортных средств
CREATE TABLE  "UP07".Vehicle (
    ID SERIAL PRIMARY KEY,
    Brand VARCHAR(30) NOT NULL,
    Model VARCHAR(30) NOT NULL,
    Plate_Number VARCHAR(10) NOT NULL,
    Year INTEGER NOT NULL,
    Category_ID INTEGER NOT NULL REFERENCES Category(ID),
    Condition VARCHAR(30) NOT NULL
);

-- Таблица Занятий
CREATE TABLE  "UP07".Lesson (
    ID SERIAL PRIMARY KEY,
    Group_ID INTEGER NOT NULL REFERENCES "Group"(ID),
    Instructor_ID INTEGER NOT NULL REFERENCES Instructor(ID),
    Vehicle_ID INTEGER REFERENCES Vehicle(ID),
    Type VARCHAR(20) NOT NULL,
    Date DATE NOT NULL,
    Time TIME NOT NULL,
    Room VARCHAR(20)
);

-- Таблица Экзаменов
CREATE TABLE  "UP07".Exam (
    ID SERIAL PRIMARY KEY,
    Student_ID INTEGER NOT NULL REFERENCES Student(ID),
    Group_ID INTEGER NOT NULL REFERENCES "Group"(ID),
    Type VARCHAR(20) NOT NULL,
    Date DATE NOT NULL,
    Result VARCHAR(20) NOT NULL
);

-- Таблица Оплат
CREATE TABLE  "UP07".Payment (
    ID SERIAL PRIMARY KEY,
    Student_ID INTEGER NOT NULL REFERENCES Student(ID),
    Amount NUMERIC(10,2) NOT NULL,
    Date DATE NOT NULL,
    Purpose VARCHAR(50) NOT NULL,
    Status VARCHAR(20) NOT NULL
);
